// Prevent Esc from closing the modal error dialog (PBI1 5.3 / PBI4 8)
errorDialog.addEventListener('cancel', (event) => {
    event.preventDefault();
});

errorDialog.addEventListener('keydown', (event) => {
    if (event.key === 'Escape') event.preventDefault();
});

function showErrorDialog() {
    if (!errorDialog.open) {
        errorDialog.showModal();
    }
}

// Keep API requests under the same team path as the current page (TR-01)
function getApiBasePrefix() {
    const teamPath = window.location.pathname.match(/^\/intproj26\/[^/]+(?=\/|$)/);
    return (teamPath ? teamPath[0] : '') + '/api/v1';
}

// Convert MONDAY -> Monday per UI Contract
function formatDay(dayStr) {
    if (!dayStr) return '';
    return dayStr.charAt(0).toUpperCase() + dayStr.slice(1).toLowerCase();
}

function createScheduleRow(item, actions = []) {
    const tr = document.createElement('tr');
    tr.setAttribute('data-testid', 'schedule-row');

    const courseObj = item.course || {};
    const cells = [
        { testid: 'code', text: courseObj.code || '' },
        { testid: 'title', text: courseObj.title || '' },
        { testid: 'type', text: courseObj.type || '' },
        { testid: 'section', text: item.section != null ? String(item.section) : '' },
        { testid: 'group', text: item.studentGroup || '' },
        { testid: 'day', text: formatDay(item.dayOfWeek) },
        { testid: 'time', text: `${item.startTime}-${item.endTime}` },
        { testid: 'weeks', text: `${item.startWeek}-${item.endWeek}` },
        { testid: 'room', text: item.room || '' },
        { testid: 'remarks', text: item.remarks || '' }
    ];

    for (const c of cells) {
        const td = document.createElement('td');
        td.setAttribute('data-testid', c.testid);
        td.textContent = c.text;
        tr.appendChild(td);
    }

    const actionCell = document.createElement('td');
    for (const [label, onAction] of actions) {
        const button = document.createElement('button');
        button.type = 'button';
        button.className = 'action-btn';
        if (label === 'Change') button.classList.add('change-btn');
        button.textContent = label;
        button.addEventListener('click', () => onAction(item));
        actionCell.appendChild(button);
    }
    tr.appendChild(actionCell);

    return tr;
}

function renderStudentTables(schedules, selections, { select, remove, change, optionsByCourse }) {
    const selectedSectionByCourse = new Map();
    for (const sel of selections) {
        selectedSectionByCourse.set(Number(sel.courseId), Number(sel.section));
    }

    const myScheduleRows = [];
    const courseOptionRows = [];

    for (const item of schedules) {
        const courseId = Number(item.course?.id);
        if (item.fixed) {
            myScheduleRows.push(item);
        } else if (selectedSectionByCourse.has(courseId)) {
            if (selectedSectionByCourse.get(courseId) === Number(item.section)) {
                myScheduleRows.push(item);
            }
        } else {
            courseOptionRows.push(item);
        }
    }

    // Render "My Schedule" table
    myScheduleTbody.innerHTML = '';
    if (myScheduleRows.length === 0) {
        myScheduleEmptyEl.hidden = false;
    } else {
        myScheduleEmptyEl.hidden = true;
        for (const row of myScheduleRows) {
            const rowActions = [];
            if (!row.fixed) {
                const sections = optionsByCourse.get(row.course.id);
                if (sections.size > 1) {
                    rowActions.push(['Change', (item) => change(item, sections)]);
                }
                rowActions.push(['Remove', remove]);
            }
            myScheduleTbody.appendChild(createScheduleRow(row, rowActions));
        }
    }

    // Render "Courses to Choose" table (hidden along with its heading when empty: PBI4 5.2)
    courseOptionsTbody.innerHTML = '';
    if (courseOptionRows.length === 0) {
        courseOptionsSection.hidden = true;
        courseOptionsHeading.hidden = true;
        courseOptionsTable.hidden = true;
    } else {
        courseOptionsSection.hidden = false;
        courseOptionsHeading.hidden = false;
        courseOptionsTable.hidden = false;
        for (const row of courseOptionRows) {
            courseOptionsTbody.appendChild(createScheduleRow(row, [['Select', select]]));
        }
    }

    studentContentEl.hidden = false;
}

เจาะลึกทีละฟังก์ชันในบล็อกนี้

  • 1. ป้องกันการปิด Error Dialog (cancel & keydown): ตามกฎ PBI 1 ข้อ 5.3 และ PBI 4 ข้อ 8 เมื่อเกิด Error ตอนโหลดข้อมูล ต้องเปิด Modal Dialog ด้วย .showModal() ที่ไม่มีปุ่มปิด และกด Esc ต้องไม่ปิด เราจึงดักทั้ง event cancel และ Escape แล้วสั่ง event.preventDefault()
  • 2. getApiBasePrefix(): ใช้ Regex ตรวจว่าเว็บเปิดอยู่ที่ path /intproj26/<TEAM-PATH> หรือไม่ เพื่อให้ยิง API ใต้ path ของทีมตัวเองเสมอทั้งตอนรันในเครื่อง (localhost) และบนเซิร์ฟเวอร์จริง (ตรงตาม TR-01)
  • 3. formatDay(dayStr): แปลงชื่อวันจาก Database ที่เป็นตัวพิมพ์ใหญ่ล้วน (MONDAY) ให้เป็นรูปแบบ UI Contract (Monday)
  • 4. createScheduleRow(item, actions): สร้างแถว <tr data-testid="schedule-row"> พร้อมคอลัมน์ข้อมูลครบ 10 ช่อง (เพิ่ม room ระหว่าง weeks กับ remarks ตาม PBI 4) ถ้าค่าใดเป็น null จะแสดงเป็นช่องว่าง
  • 5. renderStudentTables(...) (หัวใจของ PBI 4):
    • สร้าง Map เก็บคู่ courseId → section ของวิชาที่นักศึกษาเลือกไว้แล้ว เพื่อให้ค้นหาได้เร็วแบบ O(1)
    • วนลูป schedules จาก Backend เพื่อแยกเป็น 2 กอง:
      • ถ้า item.fixed === true → เข้าตาราง My Schedule (ใช้ค่า fixed จาก Backend โดยตรง หน้าเว็บไม่คำนวณเอง)
      • ถ้าไม่ใช่ fixed แต่นักศึกษาเลือกวิชานี้ไว้แล้ว → เอาเฉพาะแถวที่ section ตรงกับที่เลือกเข้า My Schedule (section อื่นของวิชานี้จะถูกข้ามไป ไม่แสดงใน Courses to Choose)
      • ถ้ายังไม่ได้เลือกวิชานี้เลย → เข้าตาราง Courses to Choose
    • รักษาการเรียงลำดับอัตโนมัติ: เนื่องจาก Backend เรียง schedules ตามลำดับ PBI 1 มาให้แล้ว การกรองด้วยลูป for...of จึงคงลำดับที่ถูกต้องไว้ทั้ง 2 ตารางโดยไม่ต้อง sort ใหม่
    • จัดการสถานะว่าง (Empty State):
      • ถ้า My Schedule ไม่มีแถวเลย (เช่น ปี 3 ที่ไม่มีวิชา fixed) → แสดงข้อความ No classes yet. Choose from Courses to Choose.
      • ถ้า Courses to Choose ไม่มีแถวเหลือ → ซ่อนทั้ง <section>, หัวข้อ <h2>, และตัวตารางตามกฎ PBI 4 ข้อ 5.2
โน้ตเผื่ออาจารย์ถาม: ใน createScheduleRow (ส่วนสร้างปุ่ม actionCell) และใน renderStudentTables (ส่วนส่งปุ่ม Select / Remove / Change) เป็นโค้ดที่เพื่อนในทีมมาต่อยอดเติมเข้าไปในฟังก์ชันที่คุณสร้างไว้ตอนทำ PBI 5, 6, 7 ครับ

async function initMySchedulePage() {
    const user = await requireSignIn();
    userNameEl.textContent = user.name || '';
    signOutBtn.addEventListener('click', signOut);
    signOutBtn.disabled = false;

    // PBI4 7: Lecturer sees the lecturer message and no profile or tables
    if (Array.isArray(user.roles) && user.roles.includes('lecturer')) {
        lecturerMessageEl.hidden = false;
        return;
    }

    const apiPrefix = getApiBasePrefix();
    const studentId = user.username;

    try {
        // 1. Read student profile (GET /api/v1/students/{studentId})
        const profileRes = await authFetch(`${apiPrefix}/students/${encodeURIComponent(studentId)}`);
        if (profileRes.status === 404) {
            // PBI4 3.2: Not a student of the BSc.IT program
            notStudentEl.hidden = false;
            return;
        }
        if (profileRes.status !== 200) {
            throw new Error(`Unexpected profile status: ${profileRes.status}`);
        }

        const profile = await profileRes.json();
        // UI Contract: Year <n> · Group <g> (middle dot U+00B7 with spaces)
        studentProfileEl.textContent = `Year ${profile.yearLevel} \u00B7 Group ${profile.studentGroup}`;
        studentProfileEl.hidden = false;

        // 2. Read year-level schedules and student selections once each (PBI4 6 / TR-06)
        const schedulesUrl = `${apiPrefix}/year-levels/${encodeURIComponent(profile.yearLevel)}/schedules?studentGroup=${encodeURIComponent(profile.studentGroup)}`;
        const selectionsUrl = `${apiPrefix}/students/${encodeURIComponent(studentId)}/selections`;

        const [schedulesRes, selectionsRes] = await Promise.all([
            authFetch(schedulesUrl),
            authFetch(selectionsUrl)
        ]);

        if (schedulesRes.status !== 200 || selectionsRes.status !== 200) {
            throw new Error('Failed to load schedules or selections');
        }

        const [schedules, selections] = await Promise.all([
            schedulesRes.json(),
            selectionsRes.json()
        ]);

        if (!Array.isArray(schedules) || !Array.isArray(selections)) {
            throw new Error('Invalid response format');
        }

        const actions = createSelectionActions(schedules, selections, selectionsUrl);
        renderStudentTables(schedules, selections, actions);
    } catch (err) {
        showErrorDialog();
    }
}

  • ขั้นที่ 3 — ดึงโปรไฟล์นักศึกษา (PBI 4 ข้อ 3 / TC-PBI4-13, 17):

  • ขั้นที่ 4 — ยิงขอตารางเรียน + วิชาที่เลือกพร้อมกัน (PBI 4 ข้อ 6 & TR-06): ใช้ Promise.all เรียกทั้ง 2 เส้นพร้อมกันอย่างละ 1 ครั้ง ตรวจว่าทั้งคู่ตอบ 200 และเป็น Array จริง ก่อนส่งไปวาดตารางผ่าน renderStudentTables()

แนวคำถามที่อาจารย์มักถามตอน Review โค้ด

คลิกเพื่อดูคำตอบ
1. ทำไมปุ่ม Sign Out ใน HTML ถึงต้องใส่ disabled ไว้ก่อน แล้วค่อยมาเปิดใน JS?

เพราะการเช็คสถานะล็อกอินกับ Keycloak (await requireSignIn()) เป็นการทำงานแบบ Asynchronous ถ้าปล่อยให้ปุ่มกดได้ตั้งแต่แรก ผู้ใช้ (หรือ Playwright test) อาจกดปุ่มก่อนที่บรรทัด addEventListener('click', signOut) จะทำงาน ทำให้กดไม่ติด จึงต้องใส่ disabled ไว้ใน HTML แล้วค่อยปลดล็อกเป็น false หลังผูก Event เสร็จครับ

2. ทำไมตอนล็อกอินเป็นอาจารย์ หรือตอนได้ 404 (ไม่ใช่นักศึกษา BSc.IT) ถึงต้องรีบ return ออกเลย?

ตามเงื่อนไขของ PBI 4 ข้อ 7 (TC-PBI4-12) และข้อ 3.2 (TC-PBI4-13) ระบุว่าห้ามหน้าเว็บยิง request ไปขอ schedules หรือ selections ต่อโดยไม่จำเป็นครับ การ return ทันทีจึงช่วยประหยัด request และผ่านเงื่อนไขการตรวจ network ของชุดเทสครับ

3. ทำไมตอนแยกตารางใน renderStudentTables() ถึงไม่ต้องเขียนฟังก์ชัน sort ใหม่?

เพราะ Backend API (GET /api/v1/year-levels/{yearLevel}/schedules) เรียงลำดับข้อมูลตามกฎ PBI 1 มาให้เรียบร้อยแล้ว เมื่อเราใช้ลูป for (const item of schedules) ไล่จากตัวแรกไปตัวสุดท้ายแล้วแยก push ลง myScheduleRows กับ courseOptionRows ลำดับของทั้งสองตารางก็จะยังคงตรงตาม PBI 1 โดยอัตโนมัติครับ

4. ทำไมต้องใช้ Promise.all ตอนโหลด schedules กับ selections?

เพราะหลังจากได้ profile.yearLevel และ profile.studentGroup แล้ว คำขอ GET /year-levels/.../schedules และ GET /students/.../selections ไม่ได้ขึ้นต่อกัน จึงยิงพร้อมกันแบบขนานด้วย Promise.all เพื่อให้หน้าเว็บโหลดเร็วขึ้น และยังคงเรียกอย่างละ 1 ครั้งตามกฎ TR-06 ครับ

5. ทำไม Error Dialog ต้องดักทั้ง event 'cancel' และ 'keydown' (Escape)?

ปกติเมื่อกด Esc บน <dialog> ที่เปิดด้วย showModal() เบราว์เซอร์จะยิงอีเวนต์ cancel เพื่อปิดหน้าต่าง แต่ในเบราว์เซอร์ยุคใหม่ (Chrome) มี Close Watcher ที่ถ้ากด Esc ซ้ำสองครั้งติดกันอาจยอมให้ปิดได้ การดักทั้ง cancel และ keydown จึงช่วยให้มั่นใจว่ากด Esc กี่ครั้ง Dialog ก็จะไม่ปิดตามเงื่อนไข PBI 1 ข้อ 5.3 / PBI 4 ข้อ 8 ครับ
